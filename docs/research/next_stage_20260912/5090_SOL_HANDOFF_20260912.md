```bash

  --output "$WORK/data"

  --data "$WORK/data" --model "$MODEL"

  --model "$MODEL" --data-manifest "$WORK/data/data_manifest.json" \
  --out "$RUN" --cosh-contract olmo_old_evq_exact \
  --cosh-table "$WORK/inputs/old_olmo_evq.npy"

"$PY" -m experiments.olmo_recovery_20260912.probe \
  --root "$RUN" --arm Cosh --execute

# 各表分别做未适配基线；不要把带 adapter 的 Native 当原模型。
"$PY" -m experiments.olmo_recovery_20260912.evaluate \
  --root "$RUN" --arm Native --base-native --split dev \
  --output "$RUN/eval/native_0_dev" --execute
"$PY" -m experiments.olmo_recovery_20260912.evaluate \
  --root "$RUN" --arm Cosh --unadapted --split dev \
  --output "$RUN/eval/cosh_0_dev" --execute
"$PY" -m experiments.olmo_recovery_20260912.evaluate \
  --root "$RUN" --arm OfficialYaRN --unadapted --split dev \
  --output "$RUN/eval/yarn_0_dev" --execute

# 单臂模板；三臂预算相同。8Mi 已由同一 schedule 保存，不重启新 schedule。
"$PY" -m experiments.olmo_recovery_20260912.train \
  --root "$RUN" --arm Cosh --until-cpt-tokens 33554432 --execute
"$PY" -m experiments.olmo_recovery_20260912.evaluate \
  --root "$RUN" --arm Cosh --checkpoint "$RUN/runs/Cosh/checkpoint-33554432" \
  --split dev --output "$RUN/eval/cosh_32mi_dev" --execute

# 断点示例；先查真实存在且匹配的 checkpoint。
"$PY" -m experiments.olmo_recovery_20260912.train \
  --root "$RUN" --arm Cosh --until-cpt-tokens 33554432 \
  --resume "$RUN/runs/Cosh/resume" --execute
```
